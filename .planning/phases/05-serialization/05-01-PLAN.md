```yaml
kind: table
schema: public
name: users
columns:
  - name: id
    type: bigint
    nullable: false
    default: "nextval('users_id_seq')"
constraints:
  primary_key:
    name: users_pkey
    columns: [id]
  unique:
    - name: users_email_key
      columns: [email]
indexes:
  - name: idx_users_created_at
    columns: [created_at]
    method: btree
partitioning: null
rls:
  enabled: false
data:
  file: data.csv
  columns: [id, email, created_at]
  row_count: 150000
  checksum: "sha256:abc123..."
```
