```go
type TableDef struct {
    ObjectHeader
    Columns      []ColumnDef
    Constraints  ConstraintsDef
    Indexes      []IndexDef
    Partitioning *PartitionDef   // nil for non-partitioned tables
    RLS          RLSDef
}

type ColumnDef struct {
    Name     string
    Type     string   // format_type()
    Nullable bool
    Default  string   // pg_get_expr(); empty if none
}

type ConstraintsDef struct {
    PrimaryKey *PrimaryKeyDef          // nil if no PK
    Unique     []UniqueConstraintDef
    Check      []CheckConstraintDef
}

type PrimaryKeyDef struct {
    Name    string
    Columns []string
}

type UniqueConstraintDef struct {
    Name       string
    Definition string   // pg_get_constraintdef e.g. "UNIQUE (email)"
}

type CheckConstraintDef struct {
    Name       string
    Definition string   // pg_get_constraintdef e.g. "CHECK (price > 0)"
}

type IndexDef struct {
    Name       string
    Method     string
    Definition string   // full CREATE INDEX from pg_get_indexdef()
}

type PartitionDef struct {
    Strategy      string   // "range", "list", "hash"
    KeyExpression string   // pg_get_partkeydef() output (added in Plan 01)
    Children      []string
}

type ForeignKeyDef struct {
    ObjectHeader                   // Kind=KindForeignKey
    SourceTable  string
    TargetSchema string
    TargetTable  string
    Definition   string            // pg_get_constraintdef() e.g. "FOREIGN KEY (col) REFERENCES other(id)"
}

type DDLGenerator interface {
    GenerateDDL(def ObjectDef) ([]string, error)
    GenerateDrop(def ObjectDef) ([]string, error)
}
```
